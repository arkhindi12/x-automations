┌─────────────────────────────────────────────────────────────────────┐
│ 🎯 CARA PAKAI                                                       │
└─────────────────────────────────────────────────────────────────────┘

  1. Buka browser → http://localhost:5001
  
  2. Klik menu "Accounts" di sidebar
  
  3. Klik tombol "Add Account"
  
  4. Pilih "Cookies Login"
  
  5. Isi form:
     - Username: username X tanpa @
     - Cookie JSON: paste dari Cookie-Editor
     - User-Agent: copy dari browser
  
  6. Klik "Test Connection"
  
  7. Kalau valid, klik "Save"

┌─────────────────────────────────────────────────────────────────────┐
│ 🛑 STOP SERVICES                                                    │
└─────────────────────────────────────────────────────────────────────┘

  Stop frontend:
    lsof -ti:5001 | xargs kill -9
  
  Stop backend:
    lsof -ti:3006 | xargs kill -9
  
  Stop PostgreSQL:
    cd "$HOME/Downloads/X automation"
    export PGDATA="$PWD/.pgdata"
    export PATH="/tmp/pgsql/bin:$PATH"
    pg_ctl -D .pgdata stop

┌─────────────────────────────────────────────────────────────────────┐
│ 🔄 RESTART SERVICES                                                 │
└─────────────────────────────────────────────────────────────────────┘

  Restart backend:
    cd "$HOME/Downloads/X automation"
    export DATABASE_URL=postgresql://postgres@localhost:5433/x_automation
    export PORT=3006
    nohup pnpm --filter @workspace/api-server run dev > .api-server.log 2>&1 &
  
  Restart frontend:
    export API_PROXY_TARGET=http://127.0.0.1:3006
    export PORT=5001
    export BASE_PATH=/
    nohup pnpm --filter @workspace/x-automation run dev > .frontend.log 2>&1 &
